% Measures the natural height of each poster column and prints it to the log,
% so main.tex's fixed-height columns can be checked without eyeballing a print.
\documentclass[a0, portrait, language=english, nocomputetimeused]{scadsai-poster-template}
\input{poster-setup.tex}

\newsavebox{\probebox}
\newlength{\probelen}
\newcommand{\measure}[2]{%
	\sbox{\probebox}{\begin{minipage}[t]{\colw}#2\end{minipage}}%
	\setlength{\probelen}{\dimexpr\ht\probebox+\dp\probebox\relax}%
	\typeout{RANKBIND-PROBE #1 = \the\probelen}%
}
% The band's \stat boxes have a fixed height (\bandh); this measures what the
% tallest of them actually needs, so \bandh can be set to fit rather than guessed.
\newcommand{\statprobe}[3]{%
	\sbox{\probebox}{\begin{minipage}[t]{0.2305\textwidth}%
		{\footnotesize\color{scadsaigrey}\MakeUppercase{#2}}\par\vspace{2mm}
		{\Huge\color{scadsaidarkblue}\bfseries #1}\par\vspace{3mm}
		{\footnotesize #3}%
	\end{minipage}}%
	\setlength{\probelen}{\dimexpr\ht\probebox+\dp\probebox\relax}%
	\typeout{RANKBIND-PROBE stat = \the\probelen}%
}
\begin{document}
\statprobe{0.326\,$\pm$\,0.072}{ranking score}{How high a molecule's true
	targets land. 1.0 = always first, 0.029 = guessing. Ordinary training on the
	same model: 0.015.}

	baseline that never sees the molecule. Published models: 0.05--0.67.}
\statprobe{0.95 $\rightarrow$ 0.63}{pooled AUC, given up}{One score over all
	pairs at once, and exactly the score the shortcut wins.}
\statprobe{7 datasets}{the problem is everywhere}{The same test on seven corpora:
	ordinary training fails the ranking question every time.}
