%===============================================================================
% Architecture schematic, drawn at exactly \colw so it needs no scaling.
% Blue = trained, grey = frozen.
%
% Everything below the two encoder boxes is positioned relative to what precedes
% it rather than at fixed coordinates: the boxes size themselves to their text,
% so absolute y values silently collide as soon as a label changes length.
% Avoid the (<number> |- <node>) form here; it does not parse as intended and
% silently shifts a whole branch sideways.
%===============================================================================
\begin{tikzpicture}[
	x=1cm, y=1cm, node distance=7mm,
	trained/.style={rounded corners=2mm, draw=scadsaiblue, line width=1.1pt,
		fill=rbtint, text width=9.4cm, align=center, inner sep=3mm,
		font=\small},
	frozen/.style={rounded corners=2mm, draw=scadsaigrey, line width=1.1pt,
		fill=white, text width=9.4cm, align=center, inner sep=3mm,
		font=\small},
	ext/.style={rounded corners=2mm, draw=scadsaiblue, line width=1.1pt,
		dash pattern=on 3mm off 2mm, fill=white, text width=9.4cm,
		align=center, inner sep=2.5mm, font=\footnotesize},
	wide/.style={rounded corners=2mm, draw=scadsaiblue, line width=1.1pt,
		fill=rbtint, text width=20.8cm, align=center, inner sep=3mm,
		font=\small},
	flow/.style={-{Latex[length=3.5mm, width=2.8mm]}, line width=1.1pt,
		draw=scadsaigrey},
]

% ---------------------------------------------------------------- encoders
\node (lenc) [frozen, anchor=north] at (5.8, 0) {%
	{\footnotesize\color{scadsaigrey}ligand SMILES}\\[1mm]
	\textbf{ChemBERTa}, 384-d};
\node (penc) [frozen, anchor=north] at (17.4, 0) {%
	{\footnotesize\color{scadsaigrey}protein sequence}\\[1mm]
	\textbf{ESM2-650M}, 1280-d};

% --------------------------------------------- residue pool, v5b extension
\node (pool) [ext, below=of penc] {learned-query attention pool over
	residues \emph{(v5b, +3,840 par.)}};

% -------------------------------------------------------------- projectors
\node (pproj) [trained, below=of pool] {\textbf{projector}, 256-d};
\path (pproj.north) ++(-11.6, 0) coordinate (lprojpos);
\node (lproj) [trained, anchor=north] at (lprojpos)
	{\textbf{projector}, 256-d};

% ------------------------------------------------------------------- head
\path (pproj.south) ++(-5.8, -0.9) coordinate (headpos);
\node (head) [wide, anchor=north] at (headpos)
	{\textbf{low-rank bilinear head}, rank 128};
\node (score) [anchor=north, font=\small] at ([yshift=-6mm] head.south)
	{score for this ligand--protein pair};
\node (legend) [anchor=north, font=\footnotesize, color=scadsaigrey]
	at ([yshift=-3mm] score.south)
	{Blue: trained (627,201 parameters).\quad Grey: frozen.};

% ------------------------------------------------------------------ arrows
\draw [flow] (lenc) -- (lproj);
\draw [flow] (penc) -- (pool);
\draw [flow] (pool) -- (pproj);
\draw [flow] (lproj) -- (lproj |- head.north);
\draw [flow] (pproj) -- (pproj |- head.north);
\draw [flow] (head) -- (score);
\end{tikzpicture}
