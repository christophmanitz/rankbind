\newlength{\colw}\setlength{\colw}{0.3133\textwidth}
\newlength{\colgap}\setlength{\colgap}{0.0300\textwidth}
% 6.7 cm: the tallest \stat box measures 6.45 cm (probe.tex reports it), so this
% is the natural height plus a little slack rather than a guessed one.
\newlength{\bandh}\setlength{\bandh}{6.7cm}
\newlength{\bandgap}\setlength{\bandgap}{0.9cm}
% 78.65 cm window − (band content + 2×0.75 cm padding) − gap
% 0.8 cm is held back so no column can sit flush against the footer rule.
\newlength{\colh}\setlength{\colh}{\dimexpr 78.65cm-\bandh-1.5cm-\bandgap-0.8cm\relax}

\definecolor{rbwarm}{HTML}{D95F02}     % the shortcut-taking models
\definecolor{rbtint}{HTML}{EDF4F8}     % pale ScaDS blue, for panels
\definecolor{rbtintwarm}{HTML}{FBF0E7}

% A tinted panel that fills the column width.
\newcommand{\panel}[2][rbtint]{%
	{\setlength{\fboxsep}{0.7cm}%
	\noindent\colorbox{#1}{\begin{minipage}{\dimexpr\linewidth-1.4cm\relax}#2\end{minipage}}}%
}
\newcommand{\figcap}[1]{\par\vspace{2.5mm}{\footnotesize\color{scadsaigrey}#1}\par}
\newcommand{\lead}[1]{{\normalsize #1}\par\vspace{4mm}}
\newcommand{\hlb}[1]{{\color{scadsaidarkblue}\textbf{#1}}}
\newcommand{\hlw}[1]{{\color{rbwarm}\textbf{#1}}}
\newcommand{\panelhead}[1]{{\color{scadsaidarkblue}\bfseries\footnotesize\MakeUppercase{#1}}\par\vspace{3mm}}

% Open Sans carries U+00B1 but not U+2192, and the template sets no matching
% math font, so $\pm$ / $\rightarrow$ come out undersized in a different face.
% The plus-minus is therefore the text glyph and the arrow is drawn. Both are
% sized in em, so they follow whatever font size they are used at.
\newcommand{\pmt}{\,\textpm\,}
\newcommand{\arr}{\,\tikz[baseline=-0.27em]{%
	\draw[-{Latex[length=0.42em, width=0.36em]}, line width=0.09em,
		line cap=round] (0,0) -- (0.9em,0);}\,}

% One headline number in the top band.
\newcommand{\stat}[3]{%
	\begin{minipage}[t][\bandh][t]{0.2305\textwidth}%
		{\footnotesize\color{scadsaigrey}\MakeUppercase{#2}}\par\vspace{2mm}
		{\Huge\color{scadsaidarkblue}\bfseries #1}\par\vspace{3mm}
		{\footnotesize #3}%
	\end{minipage}%
}

\newcommand{\headlineband}{%
	\noindent\panel{%
		\stat{0.326\pmt0.072}{ranking score}{How high a molecule's true
			targets land. 1.0 = always first, 0.029 = guessing. Ordinary
			training on the same model: 0.015.}%
		\hfill
		\stat{0.018}{overlap with the cheat sheet}{Favourite proteins shared
			with a baseline that never sees the molecule. Published models:
			0.05--0.67.}%
		\hfill
		\stat{0.95\arr0.63}{pooled AUC, given up}{One score over all
			pairs at once, and exactly the score the shortcut wins.}%
		\hfill
		\stat{7 datasets}{the problem is everywhere}{The same test on seven
			corpora: ordinary training fails the ranking question every time.}%
	}%
}
